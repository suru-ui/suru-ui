## 使い方

```tsx
import { ToggleButton } from "@suru-ui/react";
// または、
import { ToggleButton } from "@suru-ui/react/toggle-button";

<ToggleButton />;
```

## 構成

```tsx
import * as ToggleButton from "@suru-ui/react/toggle-button";

<ToggleButton.Root>
  <ToggleButton.Content />
  <ToggleButton.Pending>
    <ToggleButton.CircularProgress>
      <ToggleButton.CircularProgressTrack />
      <ToggleButton.CircularProgressFill>
        <ToggleButton.CircularProgressFillRotation />
      </ToggleButton.CircularProgressFill>
    </ToggleButton.CircularProgress>
  </ToggleButton.Pending>
</ToggleButton.Root>;
```

## API リファレンス

### ToggleButton

`ToggleButton` は `Root` のプロパティーに加えて、以下のプロパティーを受け入れます:

<TypeTable
  type={{
    children: {
      type: "React.ReactNode",
      description: "トグルボタンの内容です。Content スロットとして表示されます。",
    },
    indeterminate: {
      type: "boolean",
      description:
        "保留状態のプログレスバーを不定モードで表示するかどうかを指定します。CircularProgress スロットに渡されます。",
    },
    percentage: {
      type: "number",
      description:
        "保留状態の進行状況をパーセンテージ（0〜100）で指定します。未定義または負の値の場合は不定モードになります。CircularProgress スロットに渡されます。",
    },
    slotProps: {
      type: "{ content?: ToggleButton.Content.Props; pending?: ToggleButton.Pending.Props; circularProgress?: ToggleButton.CircularProgress.Props; circularProgressTrack?: ToggleButton.CircularProgressTrack.Props; circularProgressFill?: ToggleButton.CircularProgressFill.Props; circularProgressFillRotation?: ToggleButton.CircularProgressFillRotation.Props }",
      description:
        "各パーツに個別のプロパティーを渡すためのオブジェクトです。各パーツのプロパティーが優先されます。",
    },
  }}
/>

### Root

`ToggleButtonRoot` は [Base UI の Toggle が受け入れるプロパティー](https://base-ui.com/react/components/toggle#api-reference) に加えて、以下のプロパティーを受け入れます:

<TypeTable
  type={{
    color: {
      type: '"blue" | "green" | "grey" | "magenta" | "orange" | "purple" | "red" | "yellow"',
      default: '"grey"',
      description: "トグルボタンの色を指定します。押下状態でのみ適用されます。",
    },
    focusableWhenDisabled: {
      type: "boolean",
      description: "ボタンが無効になっている場合でも、フォーカス可能かどうかを指定します。",
    },
    outlined: {
      type: "boolean",
      default: "false",
      description: "枠線を表示するかどうかを指定します。",
    },
    pending: {
      type: "boolean",
      default: "false",
      description:
        "true の場合、ボタンの押下が無効化され、Pending スロットが表示される保留状態（ローディング中）になります。",
    },
    radius: {
      type: '"full" | "sm" | "md" | "lg" | "none"',
      default: "size",
      description:
        'トグルボタンの角丸を指定します。"none" は角を直角にし、"lg" は大きい角丸、"full" は完全な丸形（pills 状）になります。',
    },
    size: {
      type: '"sm" | "md" | "lg"',
      default: '"md"',
      description: "トグルボタンのサイズを指定します。",
    },
    squared: {
      type: "boolean",
      default: "false",
      description:
        "true の場合、幅と高さが等しくなり、アイコンボタンなどに適した正方形になります。",
    },
    variant: {
      type: '"naked" | "liquid"',
      default: '"liquid"',
      description:
        'トグルボタンの主な外観（非押下時）を指定します。"naked" は背景のないテキストのみ、"liquid" は薄い塗りつぶしになります。押下状態では常に "solid" になります。',
    },
  }}
/>
