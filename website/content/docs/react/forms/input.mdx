## 使い方

```tsx
import { Input } from "@suru-ui/react";
// または、
import { Input } from "@suru-ui/react/input";

<Input placeholder="田中 太郎" />;
```

## 構成

```tsx
import * as Input from "@suru-ui/react/input";

<Input.Root />;
```
